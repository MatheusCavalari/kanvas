// lib/core/database/database_helper.dart
import 'package:sqflite/sqflite.dart';
import 'package:path/path.dart';

class DatabaseHelper {
  DatabaseHelper._();
  static final DatabaseHelper instance = DatabaseHelper._();

  Database? _database;

  Future<Database> get database async {
    _database ??= await _initDatabase();
    return _database!;
  }

  Future<void> close() async {
    await _database?.close();
    _database = null;
  }

  Future<Database> _initDatabase() async {
    final path = join(await getDatabasesPath(), 'fitduel.db');
    return openDatabase(path, version: 1, onCreate: _onCreate);
  }

  Future<void> _onCreate(Database db, int version) async {
    await db.execute('''
      CREATE TABLE challenges (
        id TEXT PRIMARY KEY,
        title TEXT NOT NULL,
        exercise TEXT NOT NULL,
        target INTEGER NOT NULL,
        unit TEXT NOT NULL,
        duration_days INTEGER NOT NULL,
        start_date TEXT NOT NULL,
        status TEXT NOT NULL DEFAULT 'active',
        created_at TEXT NOT NULL
      )
    ''');
    await db.execute('''
      CREATE TABLE participants (
        id TEXT PRIMARY KEY,
        challenge_id TEXT NOT NULL,
        name TEXT NOT NULL,
        is_self INTEGER NOT NULL DEFAULT 0,
        FOREIGN KEY (challenge_id) REFERENCES challenges(id)
      )
    ''');
    await db.execute('''
      CREATE TABLE progress_entries (
        id TEXT PRIMARY KEY,
        challenge_id TEXT NOT NULL,
        participant_id TEXT NOT NULL,
        value REAL NOT NULL,
        date TEXT NOT NULL,
        created_at TEXT NOT NULL,
        FOREIGN KEY (challenge_id) REFERENCES challenges(id),
        FOREIGN KEY (participant_id) REFERENCES participants(id)
      )
    ''');
    await db.execute('''
      CREATE TABLE achievements (
        id TEXT PRIMARY KEY,
        type TEXT NOT NULL,
        unlocked_at TEXT NOT NULL
      )
    ''');
  }
}
```

// lib/features/challenges/domain/challenge.dart
enum ChallengeStatus { active, completed, expired }

class Challenge {
  final String id;
  final String title;
  final String exercise;
  final int target;
  final String unit;
  final int durationDays;
  final DateTime startDate;
  final ChallengeStatus status;
  final DateTime createdAt;

  const Challenge({
    required this.id,
    required this.title,
    required this.exercise,
    required this.target,
    required this.unit,
    required this.durationDays,
    required this.startDate,
    required this.status,
    required this.createdAt,
  });

  DateTime get endDate => startDate.add(Duration(days: durationDays));

  int get daysRemaining {
    final remaining = endDate.difference(DateTime.now()).inDays;
    return remaining < 0 ? 0 : remaining;
  }

  Challenge copyWith({ChallengeStatus? status}) {
    return Challenge(
      id: id, title: title, exercise: exercise, target: target,
      unit: unit, durationDays: durationDays, startDate: startDate,
      status: status ?? this.status, createdAt: createdAt,
    );
  }
}

// lib/features/challenges/data/sqlite_challenge_repository.dart
import 'package:fitduel/core/database/database_helper.dart';
import '../domain/challenge.dart';
import '../domain/challenge_repository.dart';
import 'challenge_model.dart';

class SqliteChallengeRepository implements ChallengeRepository {
  @override
  Future<void> create(Challenge challenge) async {
    final db = await DatabaseHelper.instance.database;
    await db.insert('challenges', ChallengeModel.toMap(challenge));
  }

  @override
  Future<Challenge?> getById(String id) async {
    final db = await DatabaseHelper.instance.database;
    final rows = await db.query('challenges', where: 'id = ?', whereArgs: [id]);
    if (rows.isEmpty) return null;
    return ChallengeModel.fromMap(rows.first);
  }

  @override
  Future<List<Challenge>> getAll() async {
    final db = await DatabaseHelper.instance.database;
    final rows = await db.query('challenges', orderBy: 'created_at DESC');
    return rows.map(ChallengeModel.fromMap).toList();
  }

  @override
  Future<List<Challenge>> getActive() async {
    final db = await DatabaseHelper.instance.database;
    final rows = await db.query('challenges',
        where: 'status = ?', whereArgs: ['active'], orderBy: 'created_at DESC');
    return rows.map(ChallengeModel.fromMap).toList();
  }

  @override
  Future<void> updateStatus(String id, ChallengeStatus status) async {
    final db = await DatabaseHelper.instance.database;
    await db.update('challenges', {'status': status.name},
        where: 'id = ?', whereArgs: [id]);
  }

  @override
  Future<void> delete(String id) async {
    final db = await DatabaseHelper.instance.database;
    await db.delete('challenges', where: 'id = ?', whereArgs: [id]);
  }
}
```

class SqliteProgressRepository implements ProgressRepository {
  @override
  Future<void> addParticipant(Participant participant) async {
    final db = await DatabaseHelper.instance.database;
    await db.insert('participants', ParticipantModel.toMap(participant));
  }

  @override
  Future<List<Participant>> getParticipants(String challengeId) async {
    final db = await DatabaseHelper.instance.database;
    final rows = await db.query('participants',
        where: 'challenge_id = ?', whereArgs: [challengeId]);
    return rows.map(ParticipantModel.fromMap).toList();

    final db = await DatabaseHelper.instance.database;
    final rows = await db.query('participants',
        where: 'challenge_id = ? AND is_self = 1', whereArgs: [challengeId]);
    if (rows.isEmpty) return null;
    return ParticipantModel.fromMap(rows.first);
  }

  @override

    final db = await DatabaseHelper.instance.database;
    await db.insert('progress_entries', ProgressEntryModel.toMap(entry));

    final db = await DatabaseHelper.instance.database;
    final rows = await db.query('progress_entries',
        where: 'challenge_id = ? AND participant_id = ?',
        whereArgs: [challengeId, participantId],
        orderBy: 'date DESC');
    return rows.map(ProgressEntryModel.fromMap).toList();
  }

  @override
  Future<double> getTotalProgress(String challengeId, String participantId) async {
    final db = await DatabaseHelper.instance.database;
    final result = await db.rawQuery(
      'SELECT COALESCE(SUM(value), 0) as total FROM progress_entries WHERE challenge_id = ? AND participant_id = ?',
      [challengeId, participantId],
    );
    return (result.first['total'] as num).toDouble();

// lib/features/sync/data/payload_codec.dart
import 'dart:convert';
import 'dart:io';
import '../domain/sync_payload.dart';

class PayloadCodec {
  static String encodeChallengePayload(ChallengePayload payload) {
    return _encode(payload.toJson());
  }

  static ChallengePayload decodeChallengePayload(String encoded) {
    return ChallengePayload.fromJson(_decode(encoded));
  }

  static String encodeProgressPayload(ProgressPayload payload) {
    return _encode(payload.toJson());
  }

  static ProgressPayload decodeProgressPayload(String encoded) {
    return ProgressPayload.fromJson(_decode(encoded));
  }

  static String _encode(Map<String, dynamic> data) {
    final jsonBytes = utf8.encode(jsonEncode(data));
    final compressed = gzip.encode(jsonBytes);
    return base64Url.encode(compressed);
  }

  static Map<String, dynamic> _decode(String encoded) {
    final compressed = base64Url.decode(encoded);
    final jsonBytes = gzip.decode(compressed);
    return jsonDecode(utf8.decode(jsonBytes)) as Map<String, dynamic>;
  }
}

import 'package:flutter/material.dart';
import 'package:flutter_riverpod/flutter_riverpod.dart';

import 'package:fitduel/features/sync/data/payload_codec.dart';

class SyncScreen extends ConsumerStatefulWidget {
  const SyncScreen({super.key});

  @override
  ConsumerState<SyncScreen> createState() => _SyncScreenState();
}

class _SyncScreenState extends ConsumerState<SyncScreen> {
  bool _scanning = false;

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('Sync')),
      body: _scanning ? _buildScanner() : _buildMenu(),
    );
  }

  Widget _buildMenu() {
    return Padding(
      padding: const EdgeInsets.all(24),
      child: Column(
        crossAxisAlignment: CrossAxisAlignment.stretch,
        children: [
          _ActionCard(
            icon: Icons.qr_code_scanner,
            title: 'Escanear QR',
            subtitle: 'Entrar em um desafio ou importar progresso',
            onTap: () => setState(() => _scanning = true),
          ),
          const SizedBox(height: 16),
          _ActionCard(
            icon: Icons.qr_code,
            title: 'Meu QR',
            subtitle: 'Compartilhar meu progresso em um desafio',
            onTap: () => _showMyQr(context),
          ),
        ],
      ),
    );
  }

  Widget _buildScanner() {
    return Column(
      children: [
        Expanded(
          child: MobileScanner(
            onDetect: (capture) {
              final barcode = capture.barcodes.firstOrNull;
              if (barcode?.rawValue == null) return;
              setState(() => _scanning = false);
              _handleScannedData(barcode!.rawValue!);
            },
          ),
        ),
        SafeArea(
          child: Padding(
            padding: const EdgeInsets.all(16),
            child: OutlinedButton(
              onPressed: () => setState(() => _scanning = false),
              child: const Text('Cancelar'),
            ),
          ),
        ),
      ],
    );
  }

  Future<void> _handleScannedData(String data) async {
    try {
      final challengePayload = PayloadCodec.decodeChallengePayload(data);
      await _importChallenge(challengePayload);
      return;
    } catch (_) {}

    try {
      final progressPayload = PayloadCodec.decodeProgressPayload(data);
      await _importProgress(progressPayload);
      return;
    } catch (_) {}

    if (mounted) {
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text('QR code não reconhecido')),
      );
    }
  }

  Future<void> _importChallenge(ChallengePayload payload) async {
    final challengeRepo = ref.read(challengeRepositoryProvider);
    final progressRepo = ref.read(progressRepositoryProvider);
    final existing = await challengeRepo.getById(payload.id);
    if (existing != null) {
      if (mounted) {
        ScaffoldMessenger.of(context).showSnackBar(
          const SnackBar(content: Text('Desafio já importado!')),
        );
      }
      return;
    }
    final challenge = Challenge(
      id: payload.id, title: payload.title, exercise: payload.exercise,
      target: payload.target, unit: payload.unit, durationDays: payload.durationDays,
      startDate: payload.startDate, status: ChallengeStatus.active,
      createdAt: DateTime.now(),
    );
    await challengeRepo.create(challenge);
    final uuid = const Uuid();
    await progressRepo.addParticipant(Participant(
      id: uuid.v4(), challengeId: payload.id, name: payload.participantName, isSelf: false,
    ));
    // Prompt user for their name
    if (mounted) {
      final name = await _askName();
      if (name != null && name.isNotEmpty) {
        await progressRepo.addParticipant(Participant(
          id: uuid.v4(), challengeId: payload.id, name: name, isSelf: true,
        ));
      }
    }
    ref.invalidate(activeChallengesProvider);
    if (mounted) {
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(content: Text('Desafio "${payload.title}" importado!')),
      );
    }
  }

  Future<String?> _askName() {
    final controller = TextEditingController();
    return showDialog<String>(
      context: context,
      builder: (ctx) => AlertDialog(
        title: const Text('Seu nome'),
        content: TextField(controller: controller, autofocus: true),
        actions: [
          TextButton(onPressed: () => Navigator.pop(ctx), child: const Text('Cancelar')),
          FilledButton(
            onPressed: () => Navigator.pop(ctx, controller.text),
            child: const Text('OK'),
          ),
        ],
      ),
    );
  }

  Future<void> _importProgress(ProgressPayload payload) async {
    final progressRepo = ref.read(progressRepositoryProvider);
    final participants = await progressRepo.getParticipants(payload.challengeId);
    var participant = participants.where((p) => p.name == payload.participant).firstOrNull;
    if (participant == null) {
      participant = Participant(
        id: const Uuid().v4(), challengeId: payload.challengeId,
        name: payload.participant, isSelf: false,
      );
      await progressRepo.addParticipant(participant);
    }
    for (final e in payload.entries) {
      await progressRepo.addEntry(
        ProgressEntry(
          id: const Uuid().v4(), challengeId: payload.challengeId,
          participantId: participant.id, value: e.value, date: e.date,
          createdAt: DateTime.now(),
        ),
      );
    }
    ref.invalidate(challengeRankingProvider(payload.challengeId));
    if (mounted) {
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(content: Text('Progresso de ${payload.participant} importado!')),
      );
    }
  }

  Future<void> _showMyQr(BuildContext context) async {
    final challenges = await ref.read(challengeRepositoryProvider).getActive();
    if (challenges.isEmpty) {
      if (mounted) {
        ScaffoldMessenger.of(context).showSnackBar(
          const SnackBar(content: Text('Nenhum desafio ativo')),
        );
      }
      return;
    }
    if (!mounted) return;
    final selected = await showDialog<Challenge>(
      context: context,
      builder: (ctx) => SimpleDialog(
        title: const Text('Escolha o desafio'),
        children: challenges
            .map((c) => SimpleDialogOption(
                  child: Text(c.title),
                  onPressed: () => Navigator.pop(ctx, c),
                ))
            .toList(),
      ),
    );
    if (selected == null || !mounted) return;

    final progressRepo = ref.read(progressRepositoryProvider);
    final self = await progressRepo.getSelfParticipant(selected.id);
    if (self == null) return;
    final entries = await progressRepo.getEntries(selected.id, self.id);
    final payload = ProgressPayload(
      version: 1,
      challengeId: selected.id,
      participant: self.name,
      entries: entries
          .map((e) => ProgressPayloadEntry(date: e.date, value: e.value))
          .toList(),
    );
    final encoded = PayloadCodec.encodeProgressPayload(payload);
    if (!mounted) return;
    showModalBottomSheet(
      context: context,
      builder: (_) => Padding(
        padding: const EdgeInsets.all(32),
        child: Column(
          mainAxisSize: MainAxisSize.min,
          children: [
            Text('Meu progresso: ${selected.title}',
                style: Theme.of(context).textTheme.titleMedium),
            const SizedBox(height: 24),
            Container(
              padding: const EdgeInsets.all(16),
              decoration: BoxDecoration(
                color: Colors.white,
                borderRadius: BorderRadius.circular(12),
              ),
              child: QrImageView(data: encoded, version: QrVersions.auto, size: 200),
            ),
          ],
        ),
      ),
    );
  }
}
